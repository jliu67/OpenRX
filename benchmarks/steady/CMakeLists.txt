cmake_minimum_required(VERSION 3.16)
project(openrx_steady LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Initial steady adapter supports Linux only")
endif()
find_package(Threads REQUIRED)
get_filename_component(OPENRX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RANDOMX_LIBRARY "${OPENRX_ROOT}/build/randomx/librandomx.a" CACHE FILEPATH "Built pinned RandomX library")
if(NOT EXISTS "${RANDOMX_LIBRARY}")
  message(FATAL_ERROR "Build pinned RandomX first: bash scripts/build.sh")
endif()
add_executable(openrx-steady main.cpp)
target_compile_features(openrx-steady PRIVATE cxx_std_17)
target_compile_options(openrx-steady PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
target_include_directories(openrx-steady PRIVATE "${OPENRX_ROOT}/vendor/randomx/src")
target_link_libraries(openrx-steady PRIVATE "${RANDOMX_LIBRARY}" Threads::Threads)

def need(v,message):
    if not v:raise RuntimeError(message)


def sha(raw):return hashlib.sha256(raw).hexdigest()
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def identity(st):return [os.major(st.st_dev),os.minor(st.st_dev),st.st_ino]
def reference(p):p=Path(p);raw=p.read_bytes();return dict(path=str(p),bytes=len(raw),sha256=sha(raw))


def durable(name,raw):
    fd=os.open(HERE/name,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    try:
        with os.fdopen(fd,'wb',closefd=True)as stream:stream.write(raw);stream.flush();os.fsync(stream.fileno())
        directory=os.open(HERE,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
        try:os.fsync(directory)
        finally:os.close(directory)
    except BaseException:raise


def record(name,value):durable(name,(json.dumps(value,sort_keys=True,indent=2)+'\n').encode())


def inventory():

def lock_witness(fd):
    need(identity(os.fstat(fd))==[8,3,[identity-redacted]],'original lock identity')
    matches=[]
    for line in Path('/proc/locks').read_text().splitlines():
        fields=line.split()
        if len(fields)<8 or fields[1:4]!=['FLOCK','ADVISORY','WRITE']:continue
        dev=fields[5].split(':')
        if len(dev)==3 and int(fields[4])==os.getpid() and [int(dev[0],16),int(dev[1],16),int(dev[2])]==[8,3,[identity-redacted]]:matches.append(line)
    need(len(matches)==1,'positive kernel exclusive-flock witness')
    return dict(owner_PID=os.getpid(),lock_identity=[8,3,[identity-redacted]],kernel_proc_locks_lines=matches)


def quiescence():
    need(os.uname().nodename=='[host-identity-redacted]'and os.getuid()==1000,'exact host/account')
    present=[pid for pid in OWNED_PIDS if Path('/proc',str(pid)).exists()]
    members=[];errors=[];vanished=0;scanned=0
    for path in Path('/proc').iterdir():
        if not path.name.isdecimal():continue
        try:
            raw=(path/'stat').read_text();tail=raw[raw.rfind(')')+2:].split();scanned+=1
            if int(tail[2])==PGID or int(tail[3])==PGID:members.append(dict(PID=int(path.name),state=tail[0],PGID=int(tail[2]),SID=int(tail[3])))
        except (FileNotFoundError,ProcessLookupError):vanished+=1
        except BaseException as error:errors.append(dict(PID=path.name,error=repr(error)))
    need(not present and not members and not errors,'owned PID/PGID/SID absence certain')
    need(not SCOPE.exists(),'exact owned scope cgroup absent')
    query=subprocess.run(['/usr/bin/systemctl','--user','show',UNIT,'--property=LoadState,ActiveState,SubState,ControlGroup,Result'],stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=3,check=True)
    need(not query.stderr,'unit metadata diagnostics empty')
    props=dict(line.split('=',1)for line in query.stdout.decode().splitlines())
    need(props.get('LoadState')=='not-found'and props.get('ActiveState')=='inactive'and props.get('SubState')=='dead'and props.get('ControlGroup')=='','terminal absent owned unit')
    return dict(at_UTC=now(),owned_PIDs_absent=list(OWNED_PIDS),PGID=PGID,SID=PGID,members=members,proc_errors=errors,proc_scanned=scanned,proc_vanished_during_scan=vanished,exact_scope_cgroup_absent=True,unit_properties=props)
